include(FetchContent)
set(CMAKE_CXX_STANDARD 20)
if(UNIX AND NOT APPLE)
  FetchContent_Declare(midi_alsa_headers
    URL https://codeload.github.com/alsa-project/alsa-lib/tar.gz/81d70a27136840fa8fb1bec6eb8e04fc1475a380
    URL_HASH SHA256=ba4394a76a6fd9b9b6cf94472850794c4297308061aedf73cd3680cdd90a3d31
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(midi_alsa_headers)
  set(midi_alsa_include ${CMAKE_CURRENT_BINARY_DIR}/alsa_include)
  file(MAKE_DIRECTORY ${midi_alsa_include}/alsa)
  file(GLOB midi_alsa_header_files ${midi_alsa_headers_SOURCE_DIR}/include/*.h)
  file(COPY ${midi_alsa_header_files} DESTINATION ${midi_alsa_include}/alsa)
  file(WRITE ${midi_alsa_include}/alsa/version.h "#define SND_LIB_MAJOR 1\n#define SND_LIB_MINOR 2\n#define SND_LIB_SUBMINOR 14\n#define SND_LIB_VERSION ((1<<16)|(2<<8)|14)\n#define SND_LIB_VERSION_STR \"1.2.14\"\n")
  file(READ ${midi_alsa_headers_SOURCE_DIR}/include/asoundlib-head.h midi_alsa_public)
  string(APPEND midi_alsa_public "#include <endian.h>\n")
  foreach(header asoundef version global input output error conf pcm rawmidi ump timer hwdep control mixer ump_msg seq_event seq seqmid seq_midi_event)
    string(APPEND midi_alsa_public "#include <alsa/${header}.h>\n")
  endforeach()
  file(READ ${midi_alsa_headers_SOURCE_DIR}/include/asoundlib-tail.h midi_alsa_tail)
  file(WRITE ${midi_alsa_include}/alsa/asoundlib.h "${midi_alsa_public}${midi_alsa_tail}")
  set(ALSA_INCLUDE_DIR ${midi_alsa_include} CACHE PATH "Pinned user-local ALSA headers" FORCE)
  find_library(midi_asound_runtime NAMES asound libasound.so.2 REQUIRED)
  set(ALSA_LIBRARY ${midi_asound_runtime} CACHE FILEPATH "ALSA runtime" FORCE)
endif()
foreach(feature WINUWP WINMIDI JACK PIPEWIRE NETWORK KEYBOARD UDEV BOOST)
  set(LIBREMIDI_NO_${feature} ON CACHE BOOL "J MIDI input subset" FORCE)
endforeach()
set(LIBREMIDI_DOWNLOAD_CPPWINRT OFF CACHE BOOL "WinMM subset" FORCE)

  URL https://codeload.github.com/celtera/libremidi/tar.gz/390707b5d18b590509e823386f03fa712ef6ac1b
  URL_HASH SHA256=b9c5b22b454b1abe861890f6a296930f32dd272b478b4950a83b63822cdb0697
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(wirescape_libremidi)
if(UNIX AND NOT APPLE)
  # The downloaded headers are a build-only dependency, not an installed SDK path.
  get_target_property(midi_libremidi_includes libremidi INTERFACE_INCLUDE_DIRECTORIES)
  list(REMOVE_ITEM midi_libremidi_includes "${midi_alsa_include}")
  set_property(TARGET libremidi PROPERTY INTERFACE_INCLUDE_DIRECTORIES "${midi_libremidi_includes};$<BUILD_INTERFACE:${midi_alsa_include}>")
endif()
