cmake_minimum_required(VERSION 3.20)
project(WirescapeEssentialsSubmissions VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/artifacts")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/artifacts")

add_library(wirescape_engine SHARED host/engine.cpp)
target_compile_definitions(wirescape_engine PRIVATE WS_ENGINE_BUILD)
target_include_directories(wirescape_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(wirescape_engine PRIVATE ${CMAKE_DL_LIBS})
if(MSVC)
  target_compile_options(wirescape_engine PRIVATE /W4 /permissive-)
else()
  target_compile_options(wirescape_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()

function(wirescape_add_module target)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/module.cpp" OR NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/module.json")
    message(FATAL_ERROR "${target} requires module.cpp and module.json in its submission directory")
  endif()
  add_library(${target} SHARED module.cpp)
  target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/include")
  add_custom_command(TARGET ${target} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/module.json"
      "$<TARGET_FILE_DIR:${target}>/${target}.module.json")
endfunction()

file(GLOB submission_directories CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/submissions/*")
foreach(submission_directory IN LISTS submission_directories)
  if(IS_DIRECTORY "${submission_directory}" AND EXISTS "${submission_directory}/CMakeLists.txt")
    if(NOT EXISTS "${submission_directory}/tests.cpp")
      message(FATAL_ERROR "Submission requires tests.cpp: ${submission_directory}")
    endif()
    add_subdirectory("${submission_directory}")
  endif()
endforeach()

enable_testing()
find_package(Threads REQUIRED)
foreach(submission_directory IN LISTS submission_directories)
  if(IS_DIRECTORY "${submission_directory}" AND EXISTS "${submission_directory}/tests.cpp")
    get_filename_component(submission_name "${submission_directory}" NAME)
    if(TARGET "ws_${submission_name}")
      add_executable("test_${submission_name}" "${submission_directory}/tests.cpp")
      target_include_directories("test_${submission_name}" PRIVATE "${PROJECT_SOURCE_DIR}/include")
      target_link_libraries("test_${submission_name}" PRIVATE wirescape_engine Threads::Threads)
      add_dependencies("test_${submission_name}" "ws_${submission_name}")
      add_test(NAME "${submission_name}" COMMAND "test_${submission_name}" "$<TARGET_FILE:ws_${submission_name}>")
      set_tests_properties("${submission_name}" PROPERTIES TIMEOUT 30)
    else()
      message(FATAL_ERROR "Submission ${submission_name} must define target ws_${submission_name}; tests must not be silently skipped")
    endif()
  endif()
endforeach()
